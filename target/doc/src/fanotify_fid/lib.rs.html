1//! # fanotify-fid
2//!
3//! Linux fanotify **FID (File Identifier) mode** event parser and file handle utilities.
4//!
5//! This crate fills the gap left by [`fanotify-rs`](https://crates.io/crates/fanotify-rs),
6//! which only supports non-FID (legacy) event reading.  If you pass
7//! `FAN_REPORT_FID` / `FAN_REPORT_DIR_FID` / `FAN_REPORT_NAME` to
8//! `fanotify_init`, you **must** use this crate (or equivalent code) to
9//! correctly parse the variable-length events.
10//!
11//! ## Quick start
12//!
13//! ```rust,no_run
14//! use fanotify_fid::prelude::*;
15//!
16//! // 1. Create fanotify fd (using fanotify-rs or raw syscall)
17//! let fan_fd = fanotify_fid::fanotify_init(
18//!     FAN_CLASS_NOTIF | FAN_CLOEXEC | FAN_NONBLOCK |
19//!     FAN_REPORT_FID | FAN_REPORT_DIR_FID | FAN_REPORT_NAME,
20//!     0,
21//! ).unwrap();
22//!
23//! // 2. Open mount fd for handle resolution
24//! let mount_fd = open_mount_fd("/").unwrap();
25//!
26//! // 3. Read events
27//! let mut buf = Vec::with_capacity(65536);
28//! let events = fanotify_fid::read_fid_events(
29//!     fan_fd,
30//!     &[mount_fd],
31//!     &mut buf,
32//!     None, // no persistent cache
33//! ).unwrap();
34//!
35//! for ev in &events {
36//!     println!("{:?} {:?}", ev.event_names(), ev.path);
37//! }
38//! ```
39//!
40//! ## Relationship with `fanotify-rs`
41//!
42//! This crate is **complementary**, not a replacement.  Use `fanotify-rs` for:
43//! - Safe `fanotify_init` / `fanotify_mark` wrappers
44//! - Constants if you prefer their re-exports
45//!
46//! Use `fanotify-fid` for:
47//! - FID-mode event reading and parsing (what `fanotify-rs` doesn't support)
48//! - `name_to_handle_at` / `open_by_handle_at` safe wrappers
49//! - File handle → path resolution

66/// Thin safe wrapper around `fanotify_init` (raw syscall).
67///
68/// Provided for convenience so you don't need to depend on `fanotify-rs` just
69/// for init.  Returns the raw file descriptor on success.
70pub fn fanotify_init(flags: u32, event_f_flags: u32) -> std::io::Result<i32> {
71    // SAFETY: trivially safe — just passes flags to the kernel.
72    let fd = unsafe { libc::fanotify_init(flags as libc::c_uint, event_f_flags as libc::c_uint) };
73    if fd < 0 {
74        Err(std::io::Error::last_os_error())
75    } else {
76        Ok(fd)
77    }
78}

80/// Thin safe wrapper around `fanotify_mark` (raw syscall).
81///
82/// `path` can be a `&Path`, `&str`, or anything `AsRef<OsStr>`.
83pub fn fanotify_mark<P: AsRef<std::ffi::OsStr>>(
84    fanotify_fd: i32,
85    flags: u32,
86    mask: u64,
87    dirfd: i32,
88    path: &P,
89) -> std::io::Result<()> {
90    use std::os::unix::ffi::OsStrExt;
91    let mut raw = path.as_ref().as_bytes().to_vec();
92    raw.push(0); // null-terminate
93
94    // SAFETY: trivially safe — passes validated args to the kernel.
95    let ret = unsafe {
96        libc::fanotify_mark(
97            fanotify_fd,
98            flags as libc::c_uint,
99            mask,
100            dirfd,
101            raw.as_ptr() as *const libc::c_char,
102        )
103    };
104    if ret < 0 {
105        Err(std::io::Error::last_os_error())
106    } else {
107        Ok(())
108    }
109}
