1//! Safe wrappers around `name_to_handle_at` and `open_by_handle_at`.
2//!
3//! These two syscalls are needed to convert the file handles received in
4//! fanotify FID events back into filesystem paths.

7use std::fs;
8use std::io;
9use std::os::fd::{AsRawFd, FromRawFd, OwnedFd};
10use std::path::{Path, PathBuf};

14/// Look up the file handle for a path.
15///
16/// Calls `name_to_handle_at(AT_FDCWD, path, ...)` and returns the raw file
17/// handle bytes, which can be used as a [`HandleKey`] or passed to
18/// [`open_by_handle_at`].
19///
20/// # Errors
21///
22/// Returns an `io::Error` if the path does not exist, the process lacks
23/// permission, or the kernel does not support `name_to_handle_at` (requires
24/// Linux 2.6.39+).
25///
26/// # Example
27///
28/// ```rust,no_run
29/// use fanotify_fid::handle::name_to_handle_at;
30/// use std::path::Path;
31///
32/// let key = name_to_handle_at(Path::new("/tmp")).unwrap();
33/// ```
34pub fn name_to_handle_at(path: &Path) -> io::Result<HandleKey> {
35    let c_path = CString::new(path.as_os_str().as_encoded_bytes())
36        .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "path contains null byte"))?;
37
38    // First call to determine required size (common case: 128 is plenty).
39    let mut buf = vec![0u8; 128];
40    let mut mount_id: libc::c_int = 0;
41
42    // Write handle_bytes capacity at the start of the buffer (before file_handle header).
43    let capacity = (buf.len() as u32).to_ne_bytes();
44    buf[0..4].copy_from_slice(&capacity);
45
46    // SAFETY: `name_to_handle_at` is a pure syscall; we pass a valid C string
47    // and a buffer large enough to hold the handle.
48    let ret = unsafe {
49        libc::name_to_handle_at(
50            libc::AT_FDCWD,
51            c_path.as_ptr(),
52            buf.as_mut_ptr() as *mut libc::file_handle,
53            &mut mount_id,
54            0,
55        )
56    };
57
58    if ret != 0 {
59        let err = io::Error::last_os_error();
60        // If buffer was too small, retry with the size the kernel wrote
61        if err.raw_os_error() == Some(libc::EOVERFLOW) {
62            let needed = u32::from_ne_bytes(buf[0..4].try_into().unwrap()) as usize;
63            let mut buf = vec![0u8; needed + 64];
64            let capacity = (buf.len() as u32).to_ne_bytes();
65            buf[0..4].copy_from_slice(&capacity);
66            let ret = unsafe {
67                libc::name_to_handle_at(
68                    libc::AT_FDCWD,
69                    c_path.as_ptr(),
70                    buf.as_mut_ptr() as *mut libc::file_handle,
71                    &mut mount_id,
72                    0,
73                )
74            };
75            if ret != 0 {
76                return Err(io::Error::last_os_error());
77            }
78            let handle_bytes = u32::from_ne_bytes(buf[0..4].try_into().unwrap()) as usize;
79            buf.truncate(8 + handle_bytes);
80            return Ok(buf);
81        }
82        return Err(err);
83    }
84
85    let handle_bytes = u32::from_ne_bytes(buf[0..4].try_into().unwrap()) as usize;
86    buf.truncate(8 + handle_bytes);
87    Ok(buf)
88}

90/// Open a file by its kernel file handle.
91///
92/// Calls `open_by_handle_at(mount_fd, fh_data, O_PATH)` and returns an
93/// [`OwnedFd`] for the opened file.
94///
95/// `mount_fd` must be an open file descriptor referencing a mount point on the
96/// same filesystem that originally produced the handle.  `fh_data` is the raw
97/// handle bytes from a fanotify FID info record or from [`name_to_handle_at`].
98///
99/// The returned fd is opened with `O_PATH`, so it can be used with
100/// `readlink("/proc/self/fd/N")` to recover the path, but not for I/O.
101///
102/// # Errors
103///
104/// Returns an `io::Error` if the handle is invalid, the mount fd does not
105/// belong to the right filesystem, or the file has been deleted.
106pub fn open_by_handle_at(mount_fd: i32, fh_data: &[u8]) -> io::Result<OwnedFd> {
107    if fh_data.len() < FH_HDR_SIZE {
108        return Err(io::Error::new(
109            io::ErrorKind::InvalidInput,
110            "file_handle data too short",
111        ));
112    }
113
114    // SAFETY: `open_by_handle_at` is a pure syscall; mount_fd must be a valid
115    // fd referencing a mount point on the same filesystem as the handle.
116    let fd = unsafe {
117        libc::open_by_handle_at(
118            mount_fd,
119            fh_data.as_ptr() as *mut libc::file_handle,
120            libc::O_PATH,
121        )
122    };
123
124    if fd < 0 {
125        return Err(io::Error::last_os_error());
126    }
127
128    // SAFETY: we just successfully opened fd, and OwnedFd takes ownership.
129    Ok(unsafe { OwnedFd::from_raw_fd(fd) })
130}

132/// Resolve a file handle to an absolute path by trying each mount fd.
133///
134/// Iterates through `mount_fds`, attempting [`open_by_handle_at`] on each
135/// until one succeeds.  On success, reads the path via
136/// `readlink("/proc/self/fd/{fd}")`.
137///
138/// Returns `None` if no mount fd can resolve the handle (e.g. the file was
139/// deleted, or none of the mount fds belong to the right filesystem).
140///
141/// This is a best-effort function: on a busy system, the file may be deleted
142/// between resolution and path read.
143pub fn resolve_file_handle(mount_fds: &[i32], fh_data: &[u8]) -> Option<PathBuf> {
144    if fh_data.len() < FH_HDR_SIZE {
145        return None;
146    }
147
148    for &mfd in mount_fds {
149        match open_by_handle_at(mfd, fh_data) {
150            Ok(fd) => {
151                let result = fs::read_link(format!("/proc/self/fd/{}", fd.as_raw_fd()));
152                // fd is closed by OwnedFd::drop
153                if let Ok(p) = result {
154                    return Some(p);
155                }
156            }
157            Err(_) => continue,
158        }
159    }
160
161    None
162}
