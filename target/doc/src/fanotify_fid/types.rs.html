19// ── Kernel structure definitions ──
20
21/// `struct fanotify_event_metadata` — fixed-size header present at the start
22/// of every fanotify event.
23///
24/// In FID mode, `fd` is always `FAN_NOFD` (-1).  The header is followed by
25/// zero or more [`FanInfoHeader`] records containing file handles and optional
26/// filenames.  Use `event_len` to advance to the next event in a buffer.
27#[repr(C)]
28#[derive(Debug, Clone)]
29pub struct FanMetadata {
30    /// Total byte length of this event (header + all info records).
31    /// Use this (not `META_SIZE`) to skip to the next event.
32    pub event_len: u32,
33    /// Must equal `FANOTIFY_METADATA_VERSION` (3).
34    pub vers: u8,
35    /// Reserved, do not use.
36    pub reserved: u8,
37    /// Byte offset from the start of this event to the first info record.
38    /// Typically equals `META_SIZE`.
39    pub metadata_len: u16,
40    /// Bitmask of event types (e.g. `FAN_CREATE | FAN_MODIFY`).
41    pub mask: u64,
42    /// File descriptor (always `FAN_NOFD` = -1 in FID mode).
43    pub fd: i32,
44    /// PID of the process that triggered the event.
45    pub pid: i32,
46}
47
48/// `struct fanotify_event_info_header` — type-length header that precedes
49/// each variable-length info record within a FID event.
50///
51/// After this header comes the payload:
52/// - `FID` / `DFID`: fsid (8 bytes) + file_handle (variable)
53/// - `DFID_NAME`: fsid + file_handle + null-terminated filename + padding
54#[repr(C)]
55#[derive(Debug, Clone)]
56pub struct FanInfoHeader {
57    /// Info type: one of [`FAN_EVENT_INFO_TYPE_FID`](crate::consts::FAN_EVENT_INFO_TYPE_FID),
58    /// [`DFID`](crate::consts::FAN_EVENT_INFO_TYPE_DFID), or
59    /// [`DFID_NAME`](crate::consts::FAN_EVENT_INFO_TYPE_DFID_NAME).
60    pub info_type: u8,
61    /// Padding (unused).
62    pub pad: u8,
63    /// Total byte length of this info record (header + payload).
64    pub len: u16,
65}
66
67// ── Handle type ──
68
69/// Opaque file handle key: file_handle bytes (8-byte header + variable payload)
70/// from a fanotify FID event info record.
71///
72/// Uniquely identifies a file or directory within a filesystem.  Used as a
73/// lookup key when caching handle → path mappings to recover paths for
74/// events on deleted directories.
75pub type HandleKey = Vec<u8>;
76
77// ── Parsed event ──
78
79/// A fully parsed fanotify FID event.
80///
81/// Contains the event mask, PID, the best-effort resolved path, and optionally
82/// the raw handle keys from the event's info records.
83#[derive(Debug, Clone)]
84pub struct FidEvent {
85    /// Event mask (one or more of `FAN_CREATE`, `FAN_MODIFY`, etc.).
86    pub mask: u64,
87    /// PID of the process that triggered the event.
88    pub pid: i32,
89    /// Resolved absolute path.
90    ///
91    /// This is best-effort: it may be empty if the file or its parent directory
92    /// was deleted before the path could be resolved.  Use a persistent cache
93    /// (see [`parse::resolve_with_cache`](crate::parse::resolve_with_cache)) to
94    /// recover paths in a later read cycle.
95    pub path: PathBuf,
96    /// If the event includes a `DFID_NAME` info record: the parent directory's
97    /// handle key.  Useful for caching directory → path mappings.
98    pub dfid_name_handle: Option<HandleKey>,
99    /// If the event includes a `DFID_NAME` info record: the filename within
100    /// the parent directory.
101    pub dfid_name_filename: Option<String>,
102    /// If the event includes a `FID` or `DFID` info record: the object's own
103    /// handle key.  Useful for caching the object's path for future lookups.
104    pub self_handle: Option<HandleKey>,
105}
106
107impl FidEvent {
108    /// Returns `true` if this event indicates a queue overflow.
109    pub fn is_overflow(&self) -> bool {
110        self.mask & crate::consts::FAN_Q_OVERFLOW != 0
111    }
112
113    /// Human-readable event names from the mask (e.g. `["CREATE", "MODIFY"]`).
114    pub fn event_names(&self) -> Vec<&'static str> {
115        crate::consts::mask_to_event_names(self.mask)
116    }
